1. Get the application URL by running these commands:

  NOTE: It may take a few minutes for the LoadBalancer IP/Ingress to be available.

{{- if .Values.ingress.enabled }}
  {{- if gt (len .Values.ingress.hosts) 0 }}
  Ingress:
    http{{ if gt (len .Values.ingress.tls) 0 }}s{{ end }}://{{ (index .Values.ingress.hosts 0).host }}{{ (index (index .Values.ingress.hosts 0).paths 0).path }}
  {{- else }}
  Ingress (ALB hostname):
    kubectl -n {{ .Release.Namespace }} get ingress {{ include "cs-fundamentals.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].hostname}'; echo
  {{- end }}
{{- else if eq .Values.service.type "LoadBalancer" }}
  LoadBalancer:
    export SERVICE_IP=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "cs-fundamentals.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
    echo "http://$SERVICE_IP:{{ .Values.service.port }}"
{{- else }}
  Port-forward:
    kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "cs-fundamentals.fullname" . }} 8080:{{ .Values.service.port }}
    echo "http://127.0.0.1:8080/"
{{- end }}

---

2. Validate rollout strategy and PodDisruptionBudget:

  # Check current rollout progress and ensure maxUnavailable / surge behavior
  kubectl -n {{ .Release.Namespace }} rollout status deploy/{{ include "cs-fundamentals.fullname" . }} --watch

  # Verify PDB limits (ensures at least one pod always stays up)
  kubectl -n {{ .Release.Namespace }} get pdb {{ include "cs-fundamentals.fullname" . }} -o wide

  # Inspect events for autoscaler / drain coordination (optional)
  kubectl -n {{ .Release.Namespace }} get events --sort-by=.metadata.creationTimestamp | grep -E "Disruption|Evict|Scaled"
